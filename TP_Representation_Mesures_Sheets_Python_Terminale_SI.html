4.4. Intégration de la Relation de Tendance dans le Code du Capteur

Le microcontrôleur lit une valeur numérique brute N comprise entre 0 et 1023 sur son entrée analogique A0 (N = (Umes / 5,0) × 1023). Le programme doit :

  1. Calculer la tension analogique : U_mes = N × (5.0 / 1023.0)
  2. Calculer la résistance de la CTN en kΩ : R_ctn_kohm = R_fixe × ((5.0 / U_mes) - 1.0)
  3. Appliquer directement l'équation de la courbe de tendance trouvée en 4.2 (T = a · R2 + b · R + c) pour calculer la température réelle en °C !
C++ Arduino — Code embarqué Microcontrôleur (À compléter) Arduino IDE 
const int PIN_CTN = A0;
const float R_FIXE = 10.0; // En kΩ (10 kΩ)

// TODO: Renseigner ici les coefficients [a, b, c] obtenus lors de la courbe de tendance T = f(R)
const float COEFF_A = 0.1571;  // Exemple pour R en kΩ (à adapter avec votre équation)
const float COEFF_B = -7.6605; // Exemple (à adapter avec votre équation)
const float COEFF_C = 87.1517; // Exemple (à adapter avec votre équation)

void setup() {
  Serial.begin(9600);
  Serial.println("--- Capteur de Température CTN 10k (Modèle Tendance) ---");
}

void loop() {
  int adc = analogRead(PIN_CTN); // Lecture 0 à 1023
  
  // 1. Calcul de la tension analogique U_mes (V)
  float u_mes = adc * (5.0 / 1023.0);
  
  // 2. Calcul de la résistance de la CTN en kΩ à partir de la loi du pont diviseur
  float r_ctn_kohm = R_FIXE * ((5.0 / u_mes) - 1.0);
  
  // 3. TODO: Calcul de la température en appliquant l'équation de la courbe de tendance T = f(R)
  float temp_celsius = COEFF_A * (r_ctn_kohm * r_ctn_kohm) + COEFF_B * r_ctn_kohm + COEFF_C;

  // Affichage sur le moniteur série
  Serial.print("U_mes: "); Serial.print(u_mes, 2);
  Serial.print(" V | R_ctn: "); Serial.print(r_ctn_kohm, 2);
  Serial.print(" kOhms | Temp: "); Serial.print(temp_celsius, 1);
  Serial.println(" deg C");
  
  delay(1000);
}

# Squelette de test de la fonction de conversion avec l'équation de tendance
import numpy as np

def acquisition_temperature(adc_valeur):
    VCC = 5.0
    R_FIXE_KOHM = 10.0
    
    # 1. Calcul de la tension U_mes
    u_mes = adc_valeur * (VCC / 1023.0)
    if u_mes <= 0.05: return -999.0
    
    # 2. Calcul de la résistance R_ctn en kΩ
    r_ctn_kohm = R_FIXE_KOHM * ((VCC / u_mes) - 1.0)
    
    # 3. TODO: Appliquer l'équation de votre courbe de tendance T = a*R^2 + b*R + c
    # Remplacez a, b et c par vos valeurs obtenues à la question Q10
    a = 0.1571
    b = -7.6605
    c = 87.1517
    temp_restituee = a * (r_ctn_kohm**2) + b * r_ctn_kohm + c
    
    return u_mes, r_ctn_kohm, temp_restituee
